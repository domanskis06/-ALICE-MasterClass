Uruchamiaj każdy moduł w OSOBNYM terminalu (3 okna/karty).

--------------------------------------------------------------------------------
1) BACKEND — Django
--------------------------------------------------------------------------------

cd /home/szymon/Desktop/new_alice/alice_szymon/alice-masterclass-django

# Raz na środowisko: venv + pakiety + migracje
python3 -m venv .venv-e2e
. .venv-e2e/bin/activate
pip install -r requirements.txt
python manage.py migrate --noinput
deactivate

# Codzienny start (venv aktywny):
. .venv-e2e/bin/activate
python manage.py runserver 127.0.0.1:8000

# (opcjonalnie inny venv, np. .venv — ważne żeby manage.py działał z zainstalowanymi requirements)

--------------------------------------------------------------------------------
2) APLIKACJA UCZNIA — alice-masterclass-js
--------------------------------------------------------------------------------

cd /home/szymon/Desktop/new_alice/alice_szymon/alice-masterclass-js

# Raz: npm ci (albo npm install)
npm ci

# Start dev servera (environment web → API zwykle na localhost:8000)
npm start

# Równoważnie:
#   npm run ng:serve

--------------------------------------------------------------------------------
3) APLIKACJA NAUCZYCIELA — alice-masterclass-teacher
--------------------------------------------------------------------------------

cd /home/szymon/Desktop/new_alice/alice_szymon/alice-masterclass-teacher

# Raz: npm ci
npm ci

# Start na porcie 4201 (zgodnie z package.json)
npm start

# Równoważnie:
#   npx ng serve --host 127.0.0.1 --port 4201
